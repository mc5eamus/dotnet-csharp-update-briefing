namespace Lab06;

public static class BrokenNonTransitiveDemo
{
    // TASK 5: this is intentionally wrong. NonTransitiveMiddle is not closed,
    // so matching NonTransitiveGrandchild is not enough.
    public static string BrokenName(NonTransitiveRoot value)
        => value switch
        {
            NonTransitiveLeaf => "leaf",
            NonTransitiveGrandchild => "grandchild"
        };
}
